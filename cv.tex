\cvsection{Research Experience}

\entry{2018-Present}{\textbf{Graduate Research Assistant}, Vision Lab\\
\textit{Old Dominion University, Norfolk, Virginia; advised by Dr. Khan M. Iftekharuddin}}

\entry{2025-Present}{\textit{Quantifying Learnability in Brain Tumor Growth Modeling Under Sampling Noise}
\begin{itemize}
\item Examined how sampling noise constrains learnability in PDE-based tumor growth models.
\item Proposed the Resolvable Expressive Capacity framework with eigentask decomposition to quantify reliably learnable growth directions under uncertainty.
\item Developing a Bayesian inference framework for noise-aware digital twin models of tumor evolution.
\end{itemize}}

\entry{2023-2026}{\textit{Adult-to-Pediatric Domain Shift in Brain Tumor Segmentation}
\begin{itemize}
\item Developed a multiresolution fractal deep neural network with gradient-reversal domain alignment, transferring from a large adult glioma cohort to a substantially smaller pediatric target cohort.
\item Applied Monte-Carlo dropout for voxel-level epistemic uncertainty, flagging low-confidence regions for radiologist review.
\end{itemize}}

\entry{2021-2025}{\textit{Functional Brain Network Identification in Opioid Use Disorder} \quad Project: NIH
\begin{itemize}
\item Extracted time-frequency features from resting-state fMRI BOLD signals to localize aberrant neural activity within functional networks.
\item Applied Random Forest modeling with Boruta selection and statistical analysis to identify the functional networks with the highest discriminative power for opioid use disorder against healthy controls.
\end{itemize}}

\entry{2021-2024}{\textit{Noninvasive Classification and Prognostic Modeling of High-Grade Brain Tumors: Recurrence and Radiation-Induced Necrosis} \quad Project: NIH/NIBIB
\begin{itemize}
\item Developed a computational model addressing the clinical difficulty of separating recurrent brain tumor from radiation necrosis, which present near-identical imaging features, using multiresolution radiomic features with statistically rigorous repeated random sub-sampling.
\item Achieved an area under the curve of $0.892 \pm 0.055$ in classifying recurrence against necrosis.
\item Designed a copula-based survival pipeline addressing dependent censoring in glioblastoma, reaching cross-validated AUC $\approx 0.77$ and outperforming Cox regression with clearer stratification of patient risk groups.
\end{itemize}}

\entry{2021-2023}{\textit{Uncertainty-Aware Multi-Organ Segmentation Using MRI and CT}
\begin{itemize}
\item Proposed an uncertainty quantification framework using ensemble nnU-Nets with 5-fold cross-validation, enabling detection of low-confidence predictions.
\item Quantified predictive reliability by combining total uncertainty (entropy) with epistemic uncertainty (mutual information) across ensemble models.
\item Demonstrated that uncertainty estimates reliably stratify erroneous segmentations, improve interpretability, and enable targeted clinical error analysis.
\end{itemize}}

\entry{2018-2023}{\textit{Brain Tumor Segmentation and Validation in MRI} \quad Project: NIH/NIBIB
\begin{itemize}
\item Developed UNCE-NODE, a neural ODE-enhanced U-Net for multimodal MRI brain tumor segmentation, introducing continuous-depth modeling with shared weights across integration steps.
\item Reduced trainable parameters by roughly 30\% and accelerated inference while preserving Dice accuracy.
\item Applied uncertainty quantification to assess prediction confidence, improving the reliability and interpretability of automated segmentation.
\item Participated three times in the global challenge organized by the \href{https://www.synapse.org/Synapse:syn53708126/wiki/626320}{MICCAI} society.
\end{itemize}}

\entry{2014-2016}{\textbf{Research Assistant}, Microwave \& Optical Fiber Communications Lab\\
\textit{University of Dhaka, Bangladesh; advised by Dr. Anis Ahmed}
\begin{itemize}
\item Built simulation capability, fabrication, and experimental setup for microstrip patch antenna research in the Department of Electrical \& Electronic Engineering.
